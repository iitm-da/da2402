def kpi(label, value):
    return dbc.Card(dbc.CardBody([html.P(label), html.H3(value)]))

def graph(fig, height=420):
    fig.update_layout(height=height, margin=dict(l=40, r=20, t=50, b=40))
    return dcc.Graph(figure=fig)
